┌─────────────────────────────────────────────────────────────────────────────┐
│ API ENDPOINTS SUMMARY                                                       │
└─────────────────────────────────────────────────────────────────────────────┘

HEALTH CHECK
  GET /health
    └─ Response: { status, version, environment }

AUTHENTICATION
  POST /api/auth/register
    ├─ Body: { username, email, password }
    └─ Response: { access_token, token_type, user, disclaimer }
  
  POST /api/auth/login
    ├─ Body: { username, password }
    └─ Response: { access_token, token_type, user, disclaimer }

TRADES (All require Authorization Bearer token)
  POST /api/trades/
    ├─ Body: { pair, trade_type, amount, entry_price, stop_loss, take_profit }
    └─ Response: TradeResponse (with disclaimer)
  
  GET /api/trades/
    └─ Response: [TradeResponse] (all trades)
  
  GET /api/trades/open
    └─ Response: [TradeResponse] (only open trades)
  
  GET /api/trades/{trade_id}
    └─ Response: TradeResponse
  
  POST /api/trades/{trade_id}/close
    ├─ Query: exit_price
    └─ Response: TradeResponse (with P&L calculated)

┌─────────────────────────────────────────────────────────────────────────────┐
│ DATABASE SCHEMA                                                             │
└─────────────────────────────────────────────────────────────────────────────┘

USERS TABLE
  ├─ id (UUID, Primary Key)
  ├─ username (String, Unique)
  ├─ email (String, Unique)
  ├─ password_hash (String)
  ├─ balance (Float, Default: 129,500 KES)
  ├─ currency (String, Default: "KES")
  ├─ is_active (Boolean)
  ├─ is_admin (Boolean)
  ├─ created_at (DateTime)
  └─ updated_at (DateTime)

TRADES TABLE
  ├─ id (UUID, Primary Key)
  ├─ user_id (UUID, Foreign Key → users.id)
  ├─ pair (String)
  ├─ trade_type (Enum: BUY, SELL)
  ├─ amount (Float)
  ├─ entry_price (Float)
  ├─ exit_price (Float, Nullable)
  ├─ stop_loss (Float)
  ├─ take_profit (Float)
  ├─ status (Enum: OPEN, CLOSED, CANCELLED)
  ├─ pnl (Float)
  ├─ pnl_percent (Float)
  ├─ created_at (DateTime)
  └─ closed_at (DateTime, Nullable)

PORTFOLIOS TABLE
  ├─ id (UUID, Primary Key)
  ├─ user_id (UUID, Foreign Key → users.id, Unique)
  ├─ total_trades (Integer)
  ├─ winning_trades (Integer)
  ├─ losing_trades (Integer)
  ├─ total_pnl (Float)
  ├─ win_rate (Float)
  ├─ average_win (Float)
  ├─ average_loss (Float)
  ├─ max_win (Float)
  ├─ max_loss (Float)
  ├─ daily_pnl (Float)
  ├─ monthly_pnl (Float)
  ├─ level (Integer)
  ├─ badge (String)
  ├─ created_at (DateTime)
  └─ updated_at (DateTime)
